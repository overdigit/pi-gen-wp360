# Enable audio (loads snd_bcm2835)
dtparam=audio=on

# Automatically load initramfs files, if found
auto_initramfs=1

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

[cm4]
disable_splash=1
arm_boost=1
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1
# Enable serial ports
dtoverlay=uart3,rs485
dtoverlay=uart4,rs485
dtoverlay=uart5,rs485
# External RTC
dtoverlay=i2c-rtc,pcf85063a,i2c_csi_dsi

[cm5]
dtoverlay=dwc2,dr_mode=host
# Enable serial ports
dtoverlay=uart2-pi5,ctsrts
dtoverlay=uart3-pi5,ctsrts
dtoverlay=uart4-pi5,ctsrts
# Limit max CPU frequency to 1500 MHz to limit thermals
arm_freq=1500
# External RTC
dtoverlay=i2c-rtc,pcf85063a,i2c_csi_dsi0

[all]
# Enable CAN port
dtoverlay=spi1-1cs
dtoverlay=mcp251xfd,spi1-0
dtparam=oscillator=40000000
dtparam=interrupt=25
dtparam=speed=1000000
# Force GPIO23 high is shutdown is completed
dtoverlay=gpio-poweroff,gpiopin=23,active_low=0
dtoverlay=disable-bt
# Boot from rt kernel
kernel=kernel8_rt.img
